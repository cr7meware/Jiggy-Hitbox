local function playSound(sound)
	task.spawn(function()
		local clone = sound:Clone()
		clone.Parent = SoundService
		clone:Play()
		clone.Ended:Wait()
		clone:Destroy()
	end)
end

local fadeTweenInfo = TweenInfo.new(0.45, Enum.EasingStyle.Quint, Enum.EasingDirection.Out)

local function fadeGuiIn()
	mainFrame.Visible = true
	playOpenSound()
	bgImage.ImageTransparency = 1
	local tween = TweenService:Create(bgImage, fadeTweenInfo, {ImageTransparency = 0})
	tween:Play()
	for _, desc in ipairs(mainFrame:GetDescendants()) do
		if desc:IsA("TextLabel") or desc:IsA("TextButton") or desc:IsA("TextBox") then
			desc.TextTransparency = 1
			TweenService:Create(desc, fadeTweenInfo, {TextTransparency = 0}):Play()
		end
		if desc:IsA("UIStroke") then
			desc.Transparency = 1
			TweenService:Create(desc, fadeTweenInfo, {Transparency = 0}):Play()
		end
	end
end

local function fadeGuiOut()
	local tween = TweenService:Create(bgImage, fadeTweenInfo, {ImageTransparency = 1})
	tween:Play()
	for _, desc in ipairs(mainFrame:GetDescendants()) do
		if desc:IsA("TextLabel") or desc:IsA("TextButton") or desc:IsA("TextBox") then
			TweenService:Create(desc, fadeTweenInfo, {TextTransparency = 1}):Play()
		end
		if desc:IsA("UIStroke") then
			TweenService:Create(desc, fadeTweenInfo, {Transparency = 1}):Play()
		end
	end
	task.wait(0.5)
	mainFrame.Visible = false
end

_G.HitboxEnabled = false
local targets = {}

local function updateTargets()
	targets = {}
	for _, plr in ipairs(Players:GetPlayers()) do
		if plr ~= LocalPlayer and plr.Character and plr.Character:FindFirstChild("HumanoidRootPart") then
			table.insert(targets, plr)
		end
	end
end

local function applyHitbox()
	local sizeNum = tonumber(sizeBox.Text) or 8
	local size = math.clamp(sizeNum, 1, 10)
	for _, plr in ipairs(targets) do
		local char = plr.Character
		if char then
			local hrp = char:FindFirstChild("HumanoidRootPart")
			if hrp then
				hrp.Size = Vector3.new(size, size, size)
				hrp.Transparency = 1
				hrp.CanCollide = false
			end
		end
	end
end

local autoweaveActive = false
local autoweaveTask = nil
local HoldDuration = 0.220

local function getRemote()
	local char = LocalPlayer.Character or LocalPlayer.CharacterAdded:Wait()
	for i = 1, 20 do
		local core = char:FindFirstChild("Core")
		if core then
			local comm = core:FindFirstChild("Communicate")
			if comm and comm:FindFirstChildOfClass("RemoteEvent") then
				return comm:FindFirstChildOfClass("RemoteEvent")
			end
		end
		task.wait()
	end
	return nil
end

local function startAutoweave()
	if autoweaveActive then return end
	autoweaveActive = true
	autoweaveTask = task.spawn(function()
		while autoweaveActive do
			local remote = getRemote()
			if remote then
				remote:FireServer("Weave", nil, true)
				task.wait(HoldDuration)
				if not autoweaveActive then break end
				remote:FireServer("Weave", nil, false)
				RunService.Heartbeat:Wait()
			else
				task.wait()
			end
		end
	end)
end

local function stopAutoweave()
	autoweaveActive = false
	if autoweaveTask then task.cancel(autoweaveTask) end
	autoweaveTask = nil
	local remote = getRemote()
	if remote then
		remote:FireServer("Weave", nil, false)
	end
end

			local char = LocalPlayer.Character
			if char then
				local humanoid = char:FindFirstChildOfClass("Humanoid")

					for _, track in pairs(humanoid:GetPlayingAnimationTracks()) do
						track:AdjustSpeed(2)
					end
				end
			end
			task.wait(0.1)
		end
	end)
end

	speedAnimTask = nil

	if char then
		local humanoid = char:FindFirstChildOfClass("Humanoid")
		if humanoid then
			for _, track in pairs(humanoid:GetPlayingAnimationTracks()) do
				track:AdjustSpeed(1)
			end
		end
	end
end

				if head then
					local billboard = char:FindFirstChild("JiggyNameESP") or Instance.new("BillboardGui")

					billboard.StudsOffset = Vector3.new(0, 3, 0)

					text.Size = UDim2.new(1, 0, 1, 0)
					text.BackgroundTransparency = 1

					text.TextScaled = false

					text.Parent = billboard

			end
		end
	end
end

local function startRainbow()
	if rainbowTask then return end
	rainbowTask = task.spawn(function()
		while true do
			local hue = (tick() % 6) / 6
			local color = Color3.fromHSV(hue, 1, 1)
			for _, plr in ipairs(Players:GetPlayers()) do
				if plr ~= LocalPlayer and plr.Character then
					local hl = plr.Character:FindFirstChild("JiggyESP")
					if hl then
						hl.OutlineColor = color
					end
					local bb = plr.Character:FindFirstChild("JiggyNameESP")
					if bb then
						local txt = bb:FindFirstChild("NameText")
						if txt then
							txt.TextColor3 = color
						end
					end
				end
			end
			task.wait(0.05)
		end
	end)
end

local function stopRainbow()
	if rainbowTask then
		task.cancel(rainbowTask)
		rainbowTask = nil
	end
end

local fpsBoosted = false

local function updateToggleStyle(btn, enabled)
	btn.Text = enabled and "ON" or "OFF"
	btn.TextColor3 = Color3.new(0, 0, 0)
	local stroke = btn:FindFirstChildOfClass("UIStroke")
	if stroke then
		stroke.Color = enabled and Color3.fromRGB(0, 255, 0) or Color3.fromRGB(255, 0, 0)
	end
end

local function toggleFPSBoost()
	fpsBoosted = not fpsBoosted
	if fpsBoosted then
		pcall(function()
			game.Lighting.GlobalShadows = false
			game.Lighting.FogEnd = 1000000
			game.Lighting.Brightness = 0
			game.Lighting.ClockTime = 14
			game.Lighting.Ambient = Color3.fromRGB(60, 60, 60)
			game.Lighting.Technology = Enum.Technology.Compatibility
			settings().Rendering.QualityLevel = 1
			settings().Graphics.QualityLevel = 1
		end)
		updateToggleStyle(fpsToggle, true)
	else
		pcall(function()
			game.Lighting.GlobalShadows = true
			game.Lighting.FogEnd = 100
			game.Lighting.Brightness = 1
			game.Lighting.ClockTime = 14
			game.Lighting.Ambient = Color3.fromRGB(128, 128, 128)
			game.Lighting.Technology = Enum.Technology.Future
			settings().Rendering.QualityLevel = 10
			settings().Graphics.QualityLevel = 10
		end)
		updateToggleStyle(fpsToggle, false)
	end
end

hitboxToggle.MouseButton1Click:Connect(function()
	playClick()
	_G.HitboxEnabled = not _G.HitboxEnabled
	updateToggleStyle(hitboxToggle, _G.HitboxEnabled)
	if not _G.HitboxEnabled then
		for _, plr in ipairs(targets) do
			local char = plr.Character
			if char then
				local hrp = char:FindFirstChild("HumanoidRootPart")
				if hrp then
					hrp.Size = Vector3.new(2, 2, 1)
					hrp.Transparency = 1
					hrp.CanCollide = true
				end
			end
		end
	end
end)

autoweaveToggle.MouseButton1Click:Connect(function()
	playClick()
	if autoweaveActive then
		stopAutoweave()
	else
		startAutoweave()
	end
	updateToggleStyle(autoweaveToggle, autoweaveActive)
end)

espToggle.MouseButton1Click:Connect(function()
	playClick()
	espEnabled = not espEnabled
	if espEnabled then
		updateESP()
		startRainbow()
	else
		removeAllESP()
		stopRainbow()
	end
	updateToggleStyle(espToggle, espEnabled)
end)

end)

displayToggle.MouseButton1Click:Connect(function()
	playClick()

	updateToggleStyle(displayToggle, displayNamesEnabled)
	if displayNamesEnabled then

		end
	else
		removeNameESP()
	end
end)

local dragging = false
local dragStart = nil
local startPos = nil

topBar.InputBegan:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1 then
		dragging = true
		dragStart = input.Position
		startPos = mainFrame.Position
	end
end)

UserInputService.InputChanged:Connect(function(input)
	if dragging and input.UserInputType == Enum.UserInputType.MouseMovement then
		local delta = input.Position - dragStart
		mainFrame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
	end
end)

UserInputService.InputEnded:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1 then
		dragging = false
	end
end)

RunService.RenderStepped:Connect(function()
	updateTargets()
	if _G.HitboxEnabled then
		applyHitbox()
	end
	if espEnabled then
		updateESP()
	end
end)

Players.PlayerAdded:Connect(function(plr)
	plr.CharacterAdded:Connect(function()
		task.wait(0.6)
		if _G.HitboxEnabled then
			local sizeNum = tonumber(sizeBox.Text) or 8
			local size = math.clamp(sizeNum, 1, 10)
			local hrp = plr.Character:FindFirstChild("HumanoidRootPart")
			if hrp then
				hrp.Size = Vector3.new(size, size, size)
				hrp.Transparency = 1
				hrp.CanCollide = false
			end
		end
	end)
end)

LocalPlayer.CharacterAdded:Connect(function()
	task.wait(0.6)
	if speedWalkEnabled then
		applyWalkSpeed()
	end
end)

UserInputService.InputBegan:Connect(function(input, gameProcessed)
	if gameProcessed then return end
	if input.KeyCode == Enum.KeyCode.M then
		if mainFrame.Visible then
			fadeGuiOut()
		else
			fadeGuiIn()
		end
	end
end)

fadeGuiIn()
